    엑셀 다운로드
    이 태그, 나중에 어떻게 가져다 쓰나요? (URL 사용법)

    동작 방식 — 서버가 2초마다 이 폴더의 모든 태그를 PLC에서 미리 읽어 메모리에 들고 있습니다. 아래 URL을 호출하면 PLC와 통신하지 않고 메모리에 있는 최신값을 즉시 돌려줍니다.

    ① 폴더 전체 값 한번에 조회

    GET http://<서버IP>:5050/api/foldertag/values?folderId=4
    {
  "success": true,
  "folderId": 4,
  "lastPollAt": "2026-09-04T05:33:07Z",
  "tags": [
    { "name": "name_1", "address": "D100", "value": 50531 }
  ]
}

    ② 태그 이름으로 값 조회 (같은 이름이 여러 폴더에 있으면 tags 배열에 여러 개 담겨서 옴)

    GET http://<서버IP>:5050/api/foldertag/value/by-name?name=태그이름
    {
  "success": true,
  "count": 1,
  "tags": [
    { "id": 326, "folderId": 4, "name": "name_4", "address": "M1020", "plcId": "MST_MITSUBISHI", "value": 0 }
  ]
}

    ③ PLC 주소로 직접 조회 (태그로 등록 안 돼 있어도 동작, 응답 형태는 ②와 동일)

    GET http://<서버IP>:5050/api/foldertag/value/by-address?plcId=MST_MITSUBISHI&address=D100
    {
  "success": true,
  "count": 1,
  "tags": [
    { "id": 323, "folderId": 4, "name": "No.1 Zone", "address": "D100", "plcId": "MST_MITSUBISHI", "value": 50531 }
  ]
}

    ④ 값 쓰기 — ⚠️ 실제 설비 PLC에 값을 내보내는 되돌릴 수 없는 동작입니다. 신중하게 사용하세요.

    GET http://<서버IP>:5050/api/foldertag/write/by-name?name=태그이름&value=123
    { "success": true, "name": "name_4", "plcId": "MST_MITSUBISHI", "address": "M1020", "type": "WORD", "value": 123 }

    실패 시에는 { "success": false, "error": "..." }. 같은 이름이 여러 폴더에 걸쳐 있으면 candidates 배열로 후보를 보여주고 쓰기는 거부합니다 — folderId를 같이 지정해서 다시 호출하세요.

    위 4개는 이미 서버에 구현되어 있어 지금 바로 호출 가능합니다. 이 관리 화면은 "값"이 아니라 태그 정의(이름/주소/PLC)만 관리합니다.

    위 API, 코드로는 어떻게 호출하나요? (jQuery / Axios 예시)

    ① 폴더 전체 값 한번에 조회

    jQuery $.ajax

    $.ajax({
  url: 'http://<서버IP>:5050/api/foldertag/values',
  method: 'GET',
  data: { folderId: 4 },
  dataType: 'json',
  success: function (res) {
    if (!res.success) { console.error('조회 실패:', res.error); return; }
    res.tags.forEach(function (tag) {
      console.log(tag.name, tag.address, '=', tag.value);
    });
  },
  error: function (xhr, status, err) { console.error('요청 자체 실패:', status, err); }
});

    Axios

    async function loadFolderValues(folderId) {
  try {
    const { data } = await axios.get('http://<서버IP>:5050/api/foldertag/values', {
      params: { folderId }
    });
    if (!data.success) throw new Error(data.error);
    return data.tags; // [{ name, address, value }, ...]
  } catch (err) {
    console.error('조회 실패:', err.message);
    return [];
  }
}

    ② 태그 이름으로 값 조회 (응답은 항상 { success, count, tags:[...] } — 같은 이름이 여러 폴더에 있으면 tags에 여러 개가 옴)

    jQuery $.ajax

    $.ajax({
  url: 'http://<서버IP>:5050/api/foldertag/value/by-name',
  method: 'GET',
  data: { name: 'name_4' },
  dataType: 'json',
  success: function (res) {
    if (!res.success) { console.error('조회 실패:', res.error); return; }
    if (res.count > 1) console.warn(`같은 이름이 ${res.count}개 폴더에 있음`);
    res.tags.forEach(function (t) {
      console.log(`[폴더 ${t.folderId}]`, t.name, t.address, '=', t.value);
    });
  }
});

    Axios

    async function getTagValueByName(name) {
  const { data } = await axios.get('http://<서버IP>:5050/api/foldertag/value/by-name', {
    params: { name }
  });
  if (!data.success) throw new Error(data.error);
  return data.tags; // [{ id, folderId, name, address, plcId, value }, ...] — 보통 1개
}

    ③ PLC 주소로 직접 조회 (태그로 등록 안 돼 있어도 동작, 응답 형태는 ②와 동일)

    jQuery $.ajax

    $.ajax({
  url: 'http://<서버IP>:5050/api/foldertag/value/by-address',
  method: 'GET',
  data: { plcId: 'MST_MITSUBISHI', address: 'D100' },
  dataType: 'json',
  success: function (res) {
    if (!res.success || res.count === 0) { console.error('조회 실패 또는 없음:', res.error); return; }
    console.log(res.tags[0].plcId, res.tags[0].address, '=', res.tags[0].value);
  }
});

    Axios

    async function getTagValueByAddress(plcId, address) {
  const { data } = await axios.get('http://<서버IP>:5050/api/foldertag/value/by-address', {
    params: { plcId, address }
  });
  if (!data.success || data.count === 0) throw new Error(data.error ?? '해당 주소의 태그 없음');
  return data.tags[0].value;
}

    ④ 값 쓰기 — ⚠️ 실제 설비 PLC에 값을 내보내는 되돌릴 수 없는 동작이라, 호출 전 확인창을 넣는 걸 권장합니다.

    jQuery $.ajax

    function writeTagValue(name, value) {
  if (!confirm(`"${name}" 태그에 값 ${value}을(를) 실제로 쓰시겠습니까?`)) return;

  $.ajax({
    url: 'http://<서버IP>:5050/api/foldertag/write/by-name',
    method: 'GET',
    data: { name: name, value: value },
    dataType: 'json',
    success: function (res) {
      if (!res.success) { alert('쓰기 실패: ' + res.error); return; }
      alert(`쓰기 성공: ${res.name} (${res.plcId} / ${res.address}) = ${res.value}`);
    },
    error: function (xhr, status, err) { alert('요청 자체 실패: ' + err); }
  });
}

    Axios

    async function writeTagValue(name, value) {
  if (!window.confirm(`"${name}" 태그에 값 ${value}을(를) 실제로 쓰시겠습니까?`)) return;

  const { data } = await axios.get('http://<서버IP>:5050/api/foldertag/write/by-name', {
    params: { name, value }
  });
  if (!data.success) throw new Error(data.error);
  console.log(`쓰기 성공: ${data.name} (${data.plcId} / ${data.address}) = ${data.value}`);
  return data; // { success, name, plcId, address, type, value }
}

    4개 API 모두 GET이라 axios는 params, jQuery는 data 옵션으로 쿼리스트링을 넘기면 됩니다. HTTP 200이어도 응답 바디의 success: false로 실패를 표현할 수 있으니, error/catch 콜백만 보지 말고 data.success를 항상 먼저 확인하세요.

      태그 이름화면/보고서에 표시할 이름 (예: "No.1 Zone")
